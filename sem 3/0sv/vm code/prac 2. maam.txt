M)	Sum of First N Numbers

#!/bin/sh
echo "Enter a number:"
read n
i=1
sum=0
while [ $i -le $n ]
do
sum=$((sum + i))
i=$((i + 1))
done
echo "Sum = $sum"

Factorial of a Number
#!/bin/sh
echo "Enter a number:"
read n
fact=1
i=1
while [ $i -le $n ]
do
fact=$((fact * i))
i=$((i + 1))
done
echo "Factorial = $fact"

O)	Multi-Level Branching using Nested if-else

#!/bin/bash
echo "Enter your marks:"
read marks
if [ $marks -ge 90 ]
then
echo "Grade: A+"
elif [ $marks -ge 80 ]
then
echo "Grade: A"
elif [ $marks -ge 70 ]
then
echo "Grade: B"
elif [ $marks -ge 60 ]
then
echo "Grade: C"
elif [ $marks -ge 40 ]
then
echo "Grade: D (Pass)"
else
echo "Grade: F (Fail)"
fi
